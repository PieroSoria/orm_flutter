cmake_minimum_required(VERSION 3.13)
project(orm_flutter LANGUAGES CXX)

set(ORM_FLUTTER_LINUX_TARGET "" CACHE STRING "Override the bundled Prisma Linux target")
if(NOT ORM_FLUTTER_LINUX_TARGET)
  if(FLUTTER_TARGET_PLATFORM STREQUAL "linux-arm64")
    set(ENGINE_ARCH "arm64")
  elseif(FLUTTER_TARGET_PLATFORM STREQUAL "linux-x64")
    set(ENGINE_ARCH "x64")
  elseif(DEFINED FLUTTER_TARGET_PLATFORM AND NOT FLUTTER_TARGET_PLATFORM STREQUAL "")
    message(FATAL_ERROR "orm_flutter supports linux-x64 and linux-arm64. Unsupported target: ${FLUTTER_TARGET_PLATFORM}")
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(ENGINE_ARCH "arm64")
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    set(ENGINE_ARCH "x64")
  else()
    message(FATAL_ERROR "orm_flutter supports Linux x64 and arm64. Unsupported architecture: ${CMAKE_SYSTEM_PROCESSOR}")
  endif()
  if(EXISTS "${CMAKE_SYSROOT}/etc/alpine-release")
    if(ENGINE_ARCH STREQUAL "arm64")
      set(ORM_FLUTTER_LINUX_TARGET "linux-musl-arm64-openssl-3.0.x")
    else()
      set(ORM_FLUTTER_LINUX_TARGET "linux-musl-openssl-3.0.x")
    endif()
  elseif(ENGINE_ARCH STREQUAL "arm64")
    set(ORM_FLUTTER_LINUX_TARGET "linux-arm64-openssl-3.0.x")
  else()
    set(ORM_FLUTTER_LINUX_TARGET "debian-openssl-3.0.x")
  endif()
endif()
set(SUPPORTED_TARGETS debian-openssl-3.0.x linux-arm64-openssl-3.0.x linux-musl-openssl-3.0.x linux-musl-arm64-openssl-3.0.x)
if(NOT ORM_FLUTTER_LINUX_TARGET IN_LIST SUPPORTED_TARGETS)
  message(FATAL_ERROR "Unsupported ORM_FLUTTER_LINUX_TARGET: ${ORM_FLUTTER_LINUX_TARGET}. Bundled engines require OpenSSL 3.")
endif()
set(ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../desktop/engines/${ORM_FLUTTER_LINUX_TARGET}")
set(orm_flutter_bundled_libraries
  "${ENGINE_DIR}/prisma-query-engine"
  "${ENGINE_DIR}/prisma-schema-engine"
  "${CMAKE_CURRENT_SOURCE_DIR}/../desktop/PRISMA_LICENSE"
  PARENT_SCOPE
)
foreach(ENGINE query-engine schema-engine)
  if(NOT EXISTS "${ENGINE_DIR}/prisma-${ENGINE}")
    message(FATAL_ERROR "orm_flutter package is missing its bundled ${ENGINE} executable for ${ORM_FLUTTER_LINUX_TARGET}.")
  endif()
endforeach()
